```

│     │  └─ decision-log.md      # Key decisions made
│     ├─ reviews/
│     │  ├─ weekly-reviews.md    # Weekly health checks
│     │  └─ pdca-reviews.md      # PDCA review cycles
│     └─ outputs/
│        └─ {deliverables}       # Project deliverables
│
├─ completed/
│  └─ {project-name}/
│     ├─ project.md              # Final state
│     ├─ retrospective.md        # Post-completion analysis
│     └─ learnings.md            # Extracted learnings for global memory
│
└─ killed/
   └─ {project-name}/
      ├─ project.md              # State at kill time
      ├─ kill-reasons.md         # Why project was killed
      └─ learnings.md            # Extracted learnings
```
